function getPageTitle(pathname: string, search: string): { section: string; title: string } {
  if (pathname === '/') {
    return { section: 'AeroKDD', title: 'Vue d’ensemble du Référentiel' }
  }
  if (pathname.startsWith('/subsystems/')) {
    const code = pathname.split('/')[2] ?? 'FAN'
    const meta = getSubsystemMeta(code)
    return { section: 'Sous-Système', title: `${meta.code} — ${meta.frenchName}` }
  }
  if (pathname.startsWith('/explore')) {
    return { section: 'Analyse', title: 'Unit Explorer (GRAPH_KDD & SCHEMA KDD)' }
  }
  if (pathname.startsWith('/top-down')) {
    return { section: 'Analyse', title: 'Moteur d’Analyse Top-Down' }
  }
  if (pathname.startsWith('/network')) {
    return { section: 'Cartographie', title: 'Global Network (10 Sous-Systèmes)' }
  }
  if (pathname.startsWith('/manager')) {
    return { section: 'Référentiel', title: 'KDD Manager' }
  }
  if (pathname.startsWith('/repositories')) {
    return search.includes('tab=requirements')
      ? { section: 'Instanciation Projet', title: 'Requirements & TRACA' }
      : { section: 'Références', title: 'Bibliothèque & Sources' }
  }
  if (pathname.startsWith('/admin')) {
    return { section: 'Maintenance', title: 'Administration & Export' }
  }
  return { section: 'AeroKDD', title: 'Design System' }
}

import { useEffect, type ReactNode } from 'react'

export interface DrawerProps {
  open: boolean
  onClose: () => void
  title: string
  subtitle?: string
  headerExtra?: ReactNode
  children: ReactNode
}

export function Drawer({ open, onClose, title, subtitle, headerExtra, children }: DrawerProps) {
  useEffect(() => {
    if (!open) return
    const onKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onClose()
    }
    window.addEventListener('keydown', onKeyDown)
    return () => window.removeEventListener('keydown', onKeyDown)
  }, [open, onClose])

  if (!open) return null

  return (
    <div
      className="fixed inset-0 z-50 flex justify-end bg-[rgb(15_23_42/0.45)] backdrop-blur-[1px]"
      role="dialog"
      aria-modal="true"
      onClick={onClose}
    >
      <aside
        className="flex h-full w-full max-w-xl flex-col bg-[var(--akdd-bg-surface)] border-l border-[var(--akdd-border-default)] shadow-[var(--akdd-shadow-lg)]"
        onClick={(e) => e.stopPropagation()}
      >
        <div className="flex items-start justify-between gap-4 border-b border-[var(--akdd-border-subtle)] bg-[var(--akdd-bg-subtle)] px-6 py-4">
          <div className="flex flex-col gap-1 min-w-0">
            <h2 className="text-base font-semibold text-[var(--akdd-text-primary)]">{title}</h2>
            {subtitle && <p className="text-xs text-[var(--akdd-text-secondary)]">{subtitle}</p>}
            {headerExtra && <div className="flex flex-wrap items-center gap-2 pt-1">{headerExtra}</div>}
          </div>
          <Button variant="ghost" size="sm" onClick={onClose} aria-label="Fermer le panneau">
            <X className="h-4 w-4" />
          </Button>
        </div>

import type { ReactNode } from 'react'

export interface TabItem<T extends string> {
  id: T
  label: string
  badge?: string | number
  icon?: ReactNode
}

export interface TabsProps<T extends string> {
  tabs: TabItem<T>[]
  activeTab: T
  onChange: (id: T) => void
}

export function Tabs<T extends string>({ tabs, activeTab, onChange }: TabsProps<T>) {
  return (
    <div
      role="tablist"
      className="inline-flex flex-wrap items-center gap-1 rounded-[var(--akdd-radius-md)] bg-[var(--akdd-slate-100)] p-1 border border-[var(--akdd-border-subtle)]"
    >
      {tabs.map((tab) => {
        const isActive = tab.id === activeTab
        return (
          <button
            key={tab.id}
            role="tab"
            type="button"
            aria-selected={isActive}
            onClick={() => onChange(tab.id)}
            className={`inline-flex items-center gap-2 rounded-[var(--akdd-radius-sm)] px-3.5 py-1.5 text-sm font-medium transition-all cursor-pointer ${
              isActive
                ? 'bg-[var(--akdd-bg-surface)] text-[var(--akdd-navy-900)] shadow-[var(--akdd-shadow-sm)]'
                : 'text-[var(--akdd-text-secondary)] hover:text-[var(--akdd-text-primary)]'
            }`}
          >
            {tab.icon && <span className="inline-flex shrink-0">{tab.icon}</span>}
            <span>{tab.label}</span>
            {tab.badge !== undefined && (
              <span
                className={`rounded-full px-1.5 py-0.5 text-[11px] font-mono ${
                  isActive
                    ? 'bg-[var(--akdd-accent-blue-soft)] text-[var(--akdd-navy-700)]'
                    : 'bg-[var(--akdd-slate-200)] text-[var(--akdd-slate-600)]'
                }`}
              >
                {tab.badge}
              </span>
            )}
          </button>
        )
      })}
    </div>
  )
}
